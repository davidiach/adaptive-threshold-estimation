"""Exact arithmetic for the separately authored refined small-bias proof."""
from fractions import Fraction as Q
from pathlib import Path
import hashlib
import json
import subprocess

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[2]
PROOF = HERE.parent / "lower/proof.tex"


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    before = sha(PROOF)
    checks = {
        "variance_constant": Q(3, 4)*Q(31, 32)**2 == Q(2883, 4096) > Q(7, 10),
        "bias_square": Q(65, 64)**2*Q(1, 4) == Q(4225, 16384),
        "M_le_3V_over_2": Q(4225, 16384) < Q(1, 2)*Q(2883, 4096),
        "active_pM": Q(7, 10)-Q(4225, 16384) > Q(2, 5),
        "active_m_component": Q(4225, 16384)*Q(5, 2) < Q(3, 2),
        "response_exception": 16*Q(1, 64) == Q(1, 4),
        "combined_exception": 64+1+Q(1, 4) <= 66,
        "threshold_distances": Q(7, 16)-Q(1, 8)==Q(3, 4)-Q(7, 16)==Q(5, 16),
        "chebyshev": Q(256, 25*256)*2 == Q(2, 25) < Q(1, 10),
        "annulus": Q(3, 4)*176 == 132,
        "product_coefficient": Q(16, 3)*513*Q(5, 2)==6840,
    }
    assert all(checks.values()), checks
    after = sha(PROOF)
    assert before == after
    result = {
        "status": "EXACT-ARITHMETIC-DIAGNOSTICS; not FORMALIZED",
        "head": subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip(),
        "proof": str(PROOF.relative_to(ROOT)).replace("\\", "/"),
        "proof_sha256_before": before, "proof_sha256_after": after,
        "script_sha256": sha(Path(__file__)), "checks": checks,
        "precision": "exact integers and Fraction; no random seeds or numerical optimizer",
        "coverage": "Local constants only; probability, measurability and limiting arguments are audited on paper",
    }
    (HERE / "small_bias_arithmetic.json").write_text(json.dumps(result, indent=2)+"\n", encoding="utf-8")
    print(f"{len(checks)} refined small-bias arithmetic checks passed; source unchanged")


if __name__ == "__main__":
    main()
